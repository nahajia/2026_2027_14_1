2.
SELECT sum(varos.nepesseg)
FROM varos
INNER JOIN megye
ON megye.id=varos.megyeid
WHERE megye.mnev="Csongrád";

3.
SELECT megye.mnev, MIN(varos.nepesseg)
FROM varos
INNER JOIN megye
ON megye.id=varos.megyeid
GROUP BY megye.mnev

from varos
where varos.nepesseg<
(SELECT varos.nepesseg
FROM varos
INNER JOIN megye
ON megye.id=varos.megyeid
WHERE megye.mnev="Hajdú-Bihar"  
ORDER BY `varos`.`nepesseg` ASC
limit 1)

START TRANSACTION;
CREATE TABLE gyoriek AS
SELECT varos.vnev, varos.nepesseg, varos.terulet
FROM varos
INNER JOIN megye
ON megye.id=varos.megyeid
WHERE megye.mnev="Győr-Moson-Sopron";
UPDATE gyoriek SET gyoriek.nepesseg = gyoriek.nepesseg * 1.15;
UPDATE gyoriek SET gyoriek.terulet = gyoriek.terulet * 1.15;
